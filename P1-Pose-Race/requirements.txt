# Versions are pinned because the obvious "just install the latest" combination
# does not work on macOS (Apple silicon):
#
#   * mediapipe 1.x dropped mp.solutions AND its PoseLandmarker graph aborts on
#     arm64 with "DrishtiMetalHelper ... Service is unavailable".
#   * mediapipe 0.10.x requires numpy<2, but opencv-python >= 4.11 requires
#     numpy>=2, so OpenCV has to be held back too.
#
# This trio is the combination that actually runs.
mediapipe==0.10.21
opencv-python==4.10.0.84
# mediapipe pulls opencv-contrib-python in as a dependency, and ITS cv2 is the
# one that ends up imported. Left unpinned it resolves to a build that wants
# numpy>=2, so it has to be held at the same version as opencv-python.
opencv-contrib-python==4.10.0.84
numpy<2

legoeducation==1.1.1

# Training the pose classifier (collect_poses.py / train_poses.py).
# Not needed just to drive the car with the built-in geometry.
scikit-learn==1.9.1
joblib==1.6.0
